(define_constants [
  (OC32REG_SP 8)   ;; R8 - Stack Pointer
  (OC32REG_OV 2)   ;; R2 - OV/DZ flag (Overflow/Divide Zero)

  (OC32REG_CF 4)   ;; R4 - CF flag (Carry)

  (OC32REG_EQ 3)   ;; R3 - EQ flag (Equal)
  (OC32REG_LT 4)   ;; R4 - LT flag (Less Than)
  (OC32REG_GT 5)   ;; R5 - GT flag (Greater Than); Enabled from OC32ISA ver 1.1

  (OC32REG_TM 6)   ;; R6 - TMP Register
  (OC32REG_LR 7)   ;; R7 - Link Register
])

; Most instructions are 4 bytes long.
(define_attr "length" "" (const_int 4))

;; -------------------------------------------------------------------------
;; nop instruction
;; -------------------------------------------------------------------------
;; OC32 does not have a Real NOP instruction.
;; We use JNZ which has no side effects.

(define_insn "nop"
  [(const_int 0)]
  ""
  "\tJNZ\t\t[R0],  R0"
[(set_attr "length" "4")])

;; -------------------------------------------------------------------------
;; Conditional move instructions
;; -------------------------------------------------------------------------

;; Mode iterator for conditional move instructions
(define_mode_iterator I [QI HI SI])

(define_insn "cmov<I:mode>_ne"
  [
    (set (match_operand:I 0 "register_operand" "=r, r") 
      (if_then_else
        (match_operand:I 1 "register_operand" "r, r")
        (match_operand:I 2 "reg_or_0_operand" "r, O")
        (match_operand:I 3 "reg_or_0_operand" "O, r")
      )
    )
  ]
  ""
  "@
    \tMOVNZ\t%0,  %2,  %1
    \tMOVZ\t%0,  %3,  %1" 
  [
    (set_attr "length" "4")
  ]
)

(define_insn "cmov<I:mode>_z"
  [
    (set (match_operand:I 0 "register_operand" "=r")
	    (if_then_else:I
	      (eq:I (match_operand:I 1 "register_operand" "r") (match_operand:I 2 "const_int_operand" "O"))
	      (match_operand:I 3 "reg_or_0_operand" "rO")
        (match_operand:I 4 "reg_or_0_operand" "rO")
      )
    )
  ]
  ""
  "\tOR\t\t%0,  R0,  %4\;\tMOVZ\t%0,  %3,  %1"
  [
    (set_attr "length" "8")
  ]
)

(define_insn "cmov<I:mode>_nz"
  [
    (set (match_operand:I 0 "register_operand" "=r")
	    (if_then_else:I
	      (ne:I (match_operand:I 1 "register_operand" "r") (match_operand:I 2 "const_int_operand" "O"))
	      (match_operand:I 3 "reg_or_0_operand" "rO")
        (match_operand:I 4 "reg_or_0_operand" "rO")
      )
    )
  ]
  ""
  "\tOR\t\t%0,  R0,  %4\;\tMOVNZ\t%0,  %3,  %1"
  [
    (set_attr "length" "8")
  ]
)

  [

(define_insn "addsi3_with_carry"
  [
    (set (match_operand:SI 0 "register_operand" "=r")
	    (plus:SI 
        (plus:SI 
          (match_operand:SI 1 "reg_or_0_operand" "rO")
			    (match_operand:SI 2 "oc32_rsimm_operand" "rI")
        )
		    (match_operand:SI 3 "register_operand" "r")
      )
    )
  ]
  ""
  "\tADDC\t%0,  %1,  %2"
  [
    (set_attr "length" "4")
  ]
)

;; -------------------------------------------------------------------------
;; DImode (64-bit) addition using ADDC
;; Split into two SImode additions with carry
;; -------------------------------------------------------------------------

(define_expand "adddi3"
  [
    (set (match_operand:DI 0 "register_operand" "")
      (plus:DI 
        (match_operand:DI 1 "register_operand" "")
        (match_operand:DI 2 "register_operand" "")
      )
    )
    (clobber (mem:BLK (scratch)))
  ]
  ""
  {
    rtx low_dest = gen_lowpart (SImode, operands[0]);
    rtx high_dest = simplify_gen_subreg (SImode, operands[0], DImode, 4);
    rtx low_op1 = gen_lowpart (SImode, operands[1]);
    rtx high_op1 = simplify_gen_subreg (SImode, operands[1], DImode, 4);
    rtx low_op2 = gen_lowpart (SImode, operands[2]);
    rtx high_op2 = simplify_gen_subreg (SImode, operands[2], DImode, 4);

    /* Low 32-bit addition with carry generation */
    emit_insn (gen_addsi3 (low_dest, low_op1, low_op2));

    rtx carry = gen_rtx_REG(SImode, OC32REG_CF);

    /* High 32-bit addition with carry */
    emit_insn (gen_addsi3_with_carry (high_dest, high_op1, high_op2, carry));
    DONE;
  }
)

;; -------------------------------------------------------------------------
;; sub with borrow
;; 1-no borrow, 0-borrow
;; -------------------------------------------------------------------------
(define_insn "subsi3" 
  [
    (set (match_operand:SI 0 "register_operand" "=r") 
 	    (minus:SI 
        (match_operand:SI 1 "reg_or_0_operand" "rO") 
 	 	    (match_operand:SI 2 "reg_or_0_operand" "rO")
      )
    )
  ] 
   "" 
   "\tSUB\t\t%0,  %1,  %2" 
  [
    (set_attr "length" "4")
  ]
) 


(define_insn "subsi3_with_borrow"
  [
    (set (match_operand:SI 0 "register_operand" "=r")
	    (minus:SI 
        (minus:SI 
          (match_operand:SI 1 "reg_or_0_operand" "rO")
			    (match_operand:SI 2 "oc32_rsimm_operand" "rI")
        )
        (and:SI 
          (not:SI (match_operand:SI 3 "register_operand" "r"))
          (const_int 1)
        )
      )
    )
  ] 
  ""
  "\tSUBB\t%0,  %1,  %2"
  [
    (set_attr "length" "4")
  ]
)

;; -------------------------------------------------------------------------
;; DImode (64-bit) subtraction using SUBB
;; Split into two SImode subtractions with borrow
;; -------------------------------------------------------------------------

(define_expand "subdi3"
  [
    (set (match_operand:DI 0 "register_operand" "")
      (minus:DI 
        (match_operand:DI 1 "register_operand" "")
        (match_operand:DI 2 "register_operand" "")
      )
    )
    (clobber (mem:BLK (scratch)))
  ]
  ""
  {
    rtx low_dest = gen_lowpart (SImode, operands[0]);
    rtx high_dest = simplify_gen_subreg (SImode, operands[0], DImode, 4);
    rtx low_op1 = gen_lowpart (SImode, operands[1]);
    rtx high_op1 = simplify_gen_subreg (SImode, operands[1], DImode, 4);
    rtx low_op2 = gen_lowpart (SImode, operands[2]);
    rtx high_op2 = simplify_gen_subreg (SImode, operands[2], DImode, 4);

    /* Low 32-bit subtraction with borrow generation */
    emit_insn (gen_subsi3 (low_dest, low_op1, low_op2));
    
    rtx borrow = gen_rtx_REG(SImode, OC32REG_CF);

    /* High 32-bit subtraction with borrow */
    emit_insn (gen_subsi3_with_borrow (high_dest, high_op1, high_op2, borrow));
    
    DONE;
  }
)

;; -------------------------------------------------------------------------
;; Logical operators
;; -------------------------------------------------------------------------
;; Set Bit of Special Register @ OC32_SRADR
(define_insn "sbsr"
  [
    (set (mem:SI (match_operand:SI 0 "oc32_sr_address_operand" ""))
      (ior:SI
        (match_dup 0)
        (match_operand:SI 1 "oc32_sbsr_operand" "")
      )
    )
  ]
  ""
  {
    unsigned HOST_WIDE_INT mask = (unsigned HOST_WIDE_INT)INTVAL (operands[1]);
    unsigned HOST_WIDE_INT pos = __builtin_ctzll (mask);
    operands[1] = gen_rtx_CONST_INT (SImode, (HOST_WIDE_INT)pos);
    return "\tSBSR\t\t%0,  %1";
  }
  [(set_attr "length" "4")]
)

;; Clear Bit of Special Register @ OC32_SRADR
(define_insn "cbsr"
  [
    (set (mem:SI (match_operand:SI 0 "oc32_sr_address_operand" ""))
      (and:SI
        (match_dup 0)
        (match_operand:SI 1 "oc32_cbsr_operand" "")
      )
    )
  ]
  ""
  {
    unsigned HOST_WIDE_INT mask = (unsigned HOST_WIDE_INT)INTVAL (operands[1]);
    unsigned HOST_WIDE_INT inv = ~mask & 0xffffffffu;
    unsigned HOST_WIDE_INT pos = __builtin_ctzll (inv);
    operands[1] = gen_rtx_CONST_INT (SImode, (HOST_WIDE_INT)pos);
    return "\tCBSR\t\t%0,  %1";
  }
  [(set_attr "length" "4")]
)

;; -------------------------------------------------------------------------
;; Double move instructions (DImode/DFmode - 64 bit)
;; -------------------------------------------------------------------------
;; DImode move - uses two SImode moves
(define_expand "movdi"
  [
    (set (match_operand:DI 0 "nonimmediate_operand" "")
	    (match_operand:DI 1 "general_operand" "")
    )
  ]
  ""
  {
    if (MEM_P (operands[0]) && !const0_operand(operands[1], DImode))
      operands[1] = force_reg (DImode, operands[1]);
  }
)

(define_insn_and_split "*movdi"
  [
    (set (match_operand:DI 0 "nonimmediate_operand" "=r,r,o,r")
	    (match_operand:DI 1 "general_operand"      " rO,o,rO,n")
    )

  ]
  "register_operand (operands[0], DImode)
   || reg_or_0_operand (operands[1], DImode)"
  "#"
  "&& 1"
  [(const_int 0)]
  {

    rtx l0 = operand_subword (operands[0], 0, 0, DImode);
    rtx l1 = operand_subword (operands[1], 0, 0, DImode);
    rtx h0 = operand_subword (operands[0], 1, 0, DImode);
    rtx h1 = operand_subword (operands[1], 1, 0, DImode);

    if (reload_completed && reg_overlap_mentioned_p (l0, h1))
      {
        gcc_assert (!reg_overlap_mentioned_p (h0, l1));
        emit_move_insn (h0, h1);
        emit_move_insn (l0, l1);
      }
    else
      {
        emit_move_insn (l0, l1);
        emit_move_insn (h0, h1);
      }

    DONE;
  }
)


(define_insn_and_split "*movdf"
  [
    (set (match_operand:DF 0 "nonimmediate_operand" "=r,r,o,r")
      (match_operand:DF 1 "general_operand"      " rO,o,rO,n")
    )
  ]
  "register_operand (operands[0], DFmode)
   || reg_or_0_operand (operands[1], DFmode)"
  "#"
  "&& 1"
  [(const_int 0)]
  {
    rtx l0 = operand_subword (operands[0], 0, 0, DFmode);
    rtx l1 = operand_subword (operands[1], 0, 0, DFmode);
    rtx h0 = operand_subword (operands[0], 1, 0, DFmode);
    rtx h1 = operand_subword (operands[1], 1, 0, DFmode);

    if (reload_completed && reg_overlap_mentioned_p (l0, h1))
      {
        gcc_assert (!reg_overlap_mentioned_p (h0, l1));
        emit_move_insn (h0, h1);
        emit_move_insn (l0, l1);
      }
    else
      {
        emit_move_insn (l0, l1);
        emit_move_insn (h0, h1);
      }
    DONE;
  }
)

;; Jump instruction with range check
;; Expand hook to select direct or indirect jump
(define_insn "jump"
  [(set (pc) (label_ref (match_operand 0 "" "")))]
  ""
  {
    if (get_attr_length (insn) == 12)
      /* indirect jump */
      return "\tMOVH\tR6,  %0\;\tOR\t\tR6,  R6,  %0\;\tJ\t\t[R6]";
    else
      /* direct jump */
      return "\tJ\t\t%0";
  }

  [
    (set (attr "length")

;; "cbranch" is a necessary instruction for GCC
;; Operand layout:
;;   operands[0] = operator
;;   operands[1] = left operand (register)
;;   operands[2] = right operand (register or immediate)
;;   operands[3] = jump target (label or register)

(define_expand "cbranchsi4"
  [
    (set (pc)
      (if_then_else
        (match_operator 0 "comparison_operator" [(match_operand:SI 1 "nonmemory_operand" "") (match_operand:SI 2 "nonmemory_operand" "")])
        (label_ref (match_operand 3 "" ""))
        (pc)
      )
    )
    (clobber (mem:BLK (scratch)))
  ]
  ""
  {
    oc32_expand_cbranch (operands);
  }
)

            (const_int 1)
          )
        )
        (set (reg:SI 3)
          (if_then_else
            (eq:SI 
              (zero_extract:SI 
                (match_operand:SI 0 "register_operand" "+r")
                (const_int 1)
                (match_operand:SI 1 "const_int_operand" "b")
              )
              (const_int 0)
            )
            (const_int 1)
            (const_int 0)
          )
        )
        (set 
          (zero_extract:SI 
            (match_dup 0)
            (const_int 1)
            (match_dup 1)
          )
          (match_dup 2)
        )
      ]
    )
    
  ]
  ""
  "\tOR\t\t%2,  R0,  1\;\tEBF\t\t%2,  %0,1,%1\;\tMBF\t\t%0,1,%1,  %2"
  [
    (set_attr "length" "12")
  ]
)

;; Block memory operations from being scheduled across frame (de)allocation.
(define_insn "frame_addsi3"
  [
    (set (match_operand:SI 0 "register_operand" "=r,r")
	    (plus:SI
	      (match_operand:SI 1 "register_operand"   "%r,r")
	      (match_operand:SI 2 "reg_or_s16_operand" " r,I")
      )
    )
    (clobber (mem:BLK (scratch)))
  ]
  "reload_completed"
  "@
  \tADD\t\t%0,  %1,  %2
  \tADD\t\t%0,  %1,  %2"
  [
    (set_attr "length" "4")
  ]
)


;; Expand epilogue as RTL
(define_expand "epilogue"
  [(return)]
  ""
  {
    oc32_expand_epilogue ();
    emit_jump_insn (gen_simple_return ());
    DONE;
  }
)

(define_expand "sibcall_epilogue"
  [(return)]
  ""
  {
    oc32_expand_epilogue ();
    /* Placing a USE of LR here, rather than as a REG_USE on the
      sibcall itself, means that LR is not unnecessarily live
      within the function itself, which would force creation of
      a stack frame.  */
    emit_insn (gen_rtx_USE (VOIDmode, gen_rtx_REG (Pmode, OC32REG_LR)));
    DONE;
  }
)

(define_expand "simple_return"
  [(parallel [(simple_return) (use (match_dup 0))])]
  ""
  {
    operands[0] = gen_rtx_REG (Pmode, OC32REG_LR);
  }
)

;; This is a placeholder, during RA, in order to create the PIC regiter.
;; We do this so that we don't unconditionally mark the LR register as
;; clobbered.  It is replaced during prologue generation with the proper
;; set_got pattern below.  This works because the set_got_tmp insn is the
;; first insn in the stream and that it isn't moved during RA.
(define_insn "set_got_tmp"
  [
    (set (match_operand:SI 0 "register_operand" "=t") (unspec_volatile:SI [(const_int 0)] UNSPECV_SET_GOT))
  ]
  ""
  {
    gcc_unreachable ();
  }
)

;; The insn to initialize the GOT.
(define_insn "set_got"
  [
    (set (match_operand:SI 0 "register_operand" "=t") (unspec:SI [(const_int 0)] UNSPEC_SET_GOT))
  ]
  ""
  {
    return (
      "\tMOVH\t%0,  GOT:_GLOBAL_OFFSET_TABLE_\"
      "\tOR\t\t%0,  %0,  GOT:_GLOBAL_OFFSET_TABLE_\"
      "\tADD\t\t%0,  %0,  R1");
  }
  [
    (set_attr "length" "12")
  ]
)
